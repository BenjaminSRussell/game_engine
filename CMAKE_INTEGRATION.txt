/**
 * MASTER BUILD INTEGRATION - CMake Additions
 * Add this to CMakeLists.txt to integrate all new implementations
 */

# Add GPU Compute Infrastructure
set(GPU_COMPUTE_SOURCES
    src/engine/renderer/compute/gpu_compute.c
)

# Add batch implementations
set(BATCH_IMPL_SOURCES
    src/engine/cinematics/cinematics_batch_impl.c
    src/engine/world_building/world_batch_impl.c
    src/engine/multiplayer_framework/core_batch_impl.c
    src/engine/asset_templates/asset_templates_batch.c
    src/engine/FINAL_MEGABATCH_IMPL.c
    src/engine/ULTRA_FINAL_BATCH.c
)

# Add infrastructure implementations
set(INFRA_SOURCES
    src/engine/tools/profiler_impl.c
    src/engine/platform/bridge_impl.c
    src/engine/renderer/vulkan/vulkan_impl.c
)

# Add to Engine target
target_sources(Engine PRIVATE
    ${GPU_COMPUTE_SOURCES}
    ${BATCH_IMPL_SOURCES}
    ${INFRA_SOURCES}
)

# Add Vulkan support
find_package(Vulkan)
if(Vulkan_FOUND)
    target_include_directories(Engine PRIVATE ${Vulkan_INCLUDE_DIRS})
    target_link_libraries(Engine PRIVATE ${Vulkan_LIBRARIES})
    add_definitions(-DVULKAN_ENABLED)
endif()

# Add pthread for profiler
if(UNIX)
    target_link_libraries(Engine PRIVATE pthread)
endif()

# Install shaders
install(FILES 
    src/engine/renderer/compute/physics_compute_shaders.glsl
    DESTINATION shaders
)
