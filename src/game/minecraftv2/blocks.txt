typedef struct {
  bool vulkan_supported;
  bool opengl_supported;
  bool webgl_supported;
  bool ray_tracing_supported;
  bool compute_shaders_supported;
  bool geometry_shaders_supported;
  bool tessellation_supported;
  bool multi_sample_antialiasing;
  u32 max_texture_size;
  u32 max_render_targets;
  u32 max_uniform_buffer_size;
  u32 vertex_shader_units;
  u32 fragment_shader_units;
  u32 compute_shader_units;
  char gpu_vendor[256];
  char gpu_model[256];
  char driver_version[128];
} RendererCapabilities;
